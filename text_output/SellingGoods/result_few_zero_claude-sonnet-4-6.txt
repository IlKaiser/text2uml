@startuml

class Customer {
    String Name
    String VATNumber
}
class Address {
    String Street
    String City
    String Type
}
class Product {
    String Name
    Int StockLevel
    Boolean Available
}
class Order {
    Date PaidDate
    Date ShippedDate
    Boolean FullyShipped
    Boolean PaidInFull
    String PaymentTiming
}
class OrderLine {
    Int QuantityOrdered
    String Status
}
class Invoice {
    Boolean PaidInFull
}
class Shipment {
    String Status
    Boolean Missing
    Boolean Damaged
}

class Aircraft {
    String Name
    String Type
    Date ManufactureYear
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}

    String TicketNumber
    Double Price
    String UpgradeWish
}
class PassengerAircraft {
    Int SeatCount
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Int SeatCount
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date EmploymentDate
    Boolean IsChiefSteward
}
class Pilot {
    String Name
    Date EmploymentDate
    String License
}
class Airport {
    String Name
    String Address
    Int RunwayCount
}

PassengerAircraft "1" -- "0..*" SeatCategory
Passenger "0..*" -- "0..*" Flight

Aircraft "0..1" -- "0..1" Airline
Flight "1" -- "0..1" Airport : Departure
Flight "1" -- "0..1" Airport : Destination

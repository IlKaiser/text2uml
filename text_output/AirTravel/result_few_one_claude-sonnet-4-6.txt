@startuml

class Aircraft {
    String Name
    String Type
    Int YearOfManufacture
    Date NextInspectionDate
}
class PassengerAircraft {
    Int NumberOfSeats
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeWished
}
class SeatCategory {
    String Name
    Boolean EntertainmentProgram
    Int NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChiefSteward
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Int NumberOfRunways
}

Aircraft <|-- PassengerAircraft
Aircraft "0..*" -- "0..1" Airline
Aircraft "1..1" -- "0..*" Flight
Aircraft "0..*" -- "0..1" Airport
Passenger "0..*" -- "0..*" Flight
Flight "1..1" -- "0..*" Ticket
Ticket "0..*" -- "1..1" SeatCategory
PassengerAircraft "1..1" -- "0..*" SeatCategory
Airline "1..1" -- "0..*" FlightAttendant
Airline "1..1" -- "0..*" Pilot
FlightAttendant "0..*" -- "0..*" Flight
Pilot "1..1" -- "0..*" Flight
Flight "0..*" -- "1..1" Airport
Flight "0..*" -- "1..1" Airport

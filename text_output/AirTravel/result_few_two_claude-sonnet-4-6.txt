}

    String TicketNumber
    Double Price

}

    String Designation

}

}

    String Name
    Date DateOfEmployment
    Boolean IsChiefSteward
}

    String Name
    Date DateOfEmployment
    String License
}

    String Name
    String Address
    Int NumberOfRunways
}

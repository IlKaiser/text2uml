@startuml

class Aircraft {
    String Name
    String Type
    Int YearOfManufacture
    Date NextInspection
}
class PassengerAircraft {
    Int NumberOfSeats
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Int NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChiefSteward
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Int NumberOfRunways
}

Aircraft <|-- PassengerAircraft
Aircraft "0..*" -- "0..1" Airline
Aircraft "0..*" -- "0..1" Airport
Airline "1" -- "0..*" FlightAttendant
Airline "1" -- "0..*" Pilot
Aircraft "1" -- "0..*" Flight
Flight "0..*" -- "1" Airport
Flight "0..*" -- "1" Airport
Flight "0..*" -- "0..*" Passenger
Passenger "1" -- "0..*" Ticket
Ticket "0..*" -- "1" SeatCategory
PassengerAircraft "1" -- "0..*" SeatCategory
Flight "0..*" -- "2..3" Pilot
Flight "0..*" -- "1..*" FlightAttendant

@enduml

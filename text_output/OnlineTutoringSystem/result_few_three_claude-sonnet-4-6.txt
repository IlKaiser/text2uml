@startuml

class Student {
    String Name
    String Email
}
class Tutor {
    String Name
    String Email
    String BankAccount
}
class Subject {
    String Name
}
class TutoringOffer {
    String ExpertiseLevel
    Double HourlyPrice
}
class Availability {
    String DayOfWeek
    Time StartTime
    Time EndTime
}
class TutoringRequest {
    Date TargetDate
    Time TargetTime
    String Level
}
class TutoringSession {
    Date SessionDate
    Time SessionTime
    Double Price
    Boolean Cancelled
    String CancelledBy
}
class Payment {
    Double Amount
    String PaymentMethod
}
class Award {
    Date Month
}

@startuml

class User {
    String UniqueIdentifier
    String Name
    String Address
    Date Birthday
}
class BusinessOwner {
    String LinkedInAccount
}
class HungryCustomer {
    String DeliveryAddress
}
class Entertainer {
    String StageName
    String ShortBio
    Double PricePer30Minutes
}
class PizzaRestaurant {
    String ZipCode
    String Address
    String PhoneNumber
    String Website
    String OpeningHours
}
class Pizza {
    String Name
    String CrustStructure
    Double Price
}
class Order {
    String ID
    DateTime DateTimePlaced
    DateTime LatestDeliveryTime
    Int NumberOfPeople
}
class EntertainmentOrder {
    String TypeOfEntertainment
    Int Duration
}
class Availability {
    String Day
}

User <|-- BusinessOwner
User <|-- HungryCustomer
User <|-- Entertainer
Order <|-- EntertainmentOrder

BusinessOwner "1..1" -- "0..*" PizzaRestaurant
PizzaRestaurant "1..1" -- "0..*" Pizza
HungryCustomer "1..1" -- "0..*" Order
Order "1..1" -- "1..*" Pizza
EntertainmentOrder "0..*" -- "1..1" Entertainer
Entertainer "0..*" -- "0..*" PizzaRestaurant
(Entertainer, PizzaRestaurant) .. Availability

@enduml

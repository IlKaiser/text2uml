@startuml

class Person {
    String FirstName
    String FamilyName
    Date BirthDate
}

}
class Member {}

    String Name
    String Address
}
class Room {
    Int RoomNumber

}

}

GroupSession "0..1" -- "0..1" Trainer
GroupSession "0..*" -- "0..*" Member
IndividualSession "0..*" -- "0..1" Member

@startuml

class Person {
  String FirstName
  String FamilyName
  Date BirthDate
}

class Trainer {
  String Diploma
}

class Member {
}

class Center {
  String Name
  String Address
}

class Room {
  int Number
  int MaxCapacity
}

abstract class Session {
  Date SessionDate
  Time StartHour
}

class IndividualSession {
}

class GroupSession {
  String Type
}

Person <|-- Trainer
Person <|-- Member
Session <|-- IndividualSession
Session <|-- GroupSession

Center "1" -- "1..*" Room
Room "1" -- "0..*" Session

Trainer "0..*" -- "1" GroupSession

Member "0..*" -- "1" IndividualSession
Member "0..*" -- "1..*" GroupSession

@enduml

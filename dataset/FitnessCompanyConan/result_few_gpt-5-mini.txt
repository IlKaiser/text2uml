@startuml

class Person {
    String FirstName
    String FamilyName
    Date BirthDate
}
class Trainer {
    String Diploma
}
class FitnessCenter {
    String Name
    String Address
}
class Room {
    Int Number
    Int Capacity
}
class Session {
    Date Date
    String StartHour
}
class GroupSession {
    String Type
}
class IndividualSession {}

Trainer --|> Person
GroupSession --|> Session
IndividualSession --|> Session

FitnessCenter "1..1" -- "1..*" Room
Room "1..1" -- "0..*" Session
Trainer "1..1" -- "0..*" GroupSession
Person "0..*" -- "0..*" GroupSession
Person "1..1" -- "0..*" IndividualSession

@enduml

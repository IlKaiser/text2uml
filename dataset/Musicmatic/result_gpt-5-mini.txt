@startuml

class Song{ String Title, Integer Year, Integer Length, String Genre, String Type }
class Artist{ String Name, Date DateOfBirth, String WebsiteURL }
class User{ String ID, String Name, String Address }
class RegularUser{ }
class BusinessUser{ String VATNumber }
class Album{ String ID, String Title }
class Track{ Integer TrackNumber }

RegularUser --|> User
BusinessUser --|> User

Artist "0..*" -- "1..1" Song
BusinessUser "0..*" -- "1..1" Song
RegularUser "0..*" -- "0..*" Song
RegularUser "0..*" -- "1..1" Album
Album "0..*" -- "0..*" RegularUser
Album "2..*" -- "1..1" Track
Track "1..1" -- "0..*" Song

@enduml

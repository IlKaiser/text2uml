@startuml
class Game {
  id: String
  playerCount: Int
}
class Player {
  id: String
}
class Deck {
  id: String
  maxCards: Int
}
class Card {
  id: String
  suit: String
}
class Suit {
  id: String
  name: String
}
class Rank {
  id: String
}
class ScoringTable {
  id: String
  roundsCount: Int
}
class Round {
  id: String
  winningPlayersCount: Int
  pointValue: Int
}
class JackRound {
  id: String
}
class SoloRound {
  id: String
  soloType: String
  soloPlayerCount: Int
}
class MandatorySoloRound {
  id: String
}
class SoloType {
  id: String
  name: String
}

Game "1..1" -- "1..*" Player
Game "1..1" -- "1..1" Deck
Game "1..1" -- "1..1" ScoringTable
Deck "1..1" -- "0..*" Card
Suit "1..1" -- "1..*" Card
Rank "1..1" -- "1..*" Card
ScoringTable "1..1" -- "0..*" Round
Round "1..1" -- "1..*" Player
SoloType "1..1" -- "1..*" SoloRound
SoloRound "1..1" -- "1..1" Player

JackRound --|> Round
SoloRound --|> Round
MandatorySoloRound --|> SoloRound

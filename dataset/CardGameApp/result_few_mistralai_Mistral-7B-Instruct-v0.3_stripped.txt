class Game {
    int ID
    int PlayersCount
    Deck deck
    ScoringTable scoringTable
}

class Deck {
    int ID
    int MaxCards
}

class Card {
    int ID
    String Suit
    int Height
}

class ScoringTable {
    int ID
    int RoundsCount
}

class Round {
    int ID
    int Points
    List<Player> WinningPlayers
    boolean JackRound
    boolean SoloRound
    boolean MandatorySoloRound
    SoloType SoloType
    Player SoloPlayer
}

class Player {
    int ID
}

class SoloType {
    String Type
}

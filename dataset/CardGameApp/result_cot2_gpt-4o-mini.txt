@startuml
class Game {
  uniqueID: String
}

class Player {
  uniqueID: String
}

class Deck {
  uniqueID: String
  cards: List<Card>
}

class Card {
  uniqueID: String
  suit: String
  heightComparisonMethod: String
}

class ScoringTable {
  uniqueID: String
}

class Round {
  uniqueID: String
  winningPlayers: List<Player>
  pointValue: Int
}

  uniqueID: String
  soloType: String
  soloPlayer: Player
}

class Solo {
  uniqueID: String
}

class SoloPlayer {
  uniqueID: String
}

Game "1..1" -- "1..1" Deck
Game "1..1" -- "1..1" ScoringTable

Round "0..*" -- "0..*" WinningPlayer
Round "0..1" -- "0..1" PointValue
SoloRound "1..1" -- "1..1" Solo
SoloRound "1..1" -- "1..1" SoloPlayer

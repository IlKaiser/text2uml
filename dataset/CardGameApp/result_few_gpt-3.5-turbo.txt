
@startuml

class Game{
    Player[] Players
    Deck GameDeck
    ScoringTable GameScoringTable
}

class Player{
    int ID
}

class Deck{
    Card[] Cards
}

class Card{
    String Suit
    int ComparisonValue
}

class ScoringTable{
    Round[] Rounds
}

class Round{
    boolean JackRound
    boolean SoloRound
    Player[] WinningPlayers
    int PointValue
}

class SoloRound{
    String Type
    Player SoloPlayer
}

class MandatorySoloRound{
    String Type
    Player SoloPlayer
}

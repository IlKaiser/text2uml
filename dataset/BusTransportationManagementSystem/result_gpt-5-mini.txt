class Driver{ String DriverID, String Name, boolean IsSick }
class Bus{ String LicencePlate, boolean InRepair }
class Route{ int Number (<=9999) }
class Shift{ String Name }
class BusRouteAssignment{ Date Day }
class DriverShiftAssignment{ Date Day }

Driver "0..*" -- "1..1" DriverShiftAssignment
Bus "0..*" -- "1..1" DriverShiftAssignment
Shift "0..*" -- "1..1" DriverShiftAssignment

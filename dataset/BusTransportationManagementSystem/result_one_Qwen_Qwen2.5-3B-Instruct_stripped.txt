class Driver {
    String Name
    Boolean OnSickLeave
}
class Bus {
    String LicencePlate
    Boolean InRepairShop
}
class Route {
    String Number
}
class BusRouteAssignment {
    String RouteNumber
    String LicencePlate
    String Shift
    Driver ID
}
class Schedule {
    String Day
    List[BusRouteAssignment] Assignments
}

```plantuml
@startuml

class User{
    String Name
    String Address
    Date Birthday
}

class BusinessOwner{
    String LinkedInAccount
}

class AppUser{
    -- User
}

class PizzaRestaurant{
    String ZipCode
    String Address
    String Phone
    String Website
    String OpeningHours
}

class Pizza{
    String Name
    String CrustStructure
    Double Price
}

class BusinessOwner -- "1..*" PizzaRestaurant
class PizzaRestaurant "1..*" -- "1..*" Pizza
class User -- "0..*" AppUser

class HungryCustomer{
    String DeliveryAddress
}

class Order{
    String ID
    Date OrderPlacementTime
    Date LatestDeliveryTime
    Int NumberOfPeople
}

class EntertainmentOrder{
    String EntertainmentType
    Int Duration
}

class SpecialOrder -- "0..1" EntertainmentOrder
class AppUser "0..*" -- "0..*" HungryCustomer
class HungryCustomer -- "1..*" Order
class Order "1..*" -- "1..*" Pizza

class Entertainer{
    String StageName
    String Bio
    Double PricePer30Minutes
}

class EntertainmentOrder -- "1..1" Entertainer
class Entertainer "0..*" -- "0..*" PizzaRestaurant

@enduml
```

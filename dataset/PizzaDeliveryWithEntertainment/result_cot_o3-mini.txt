@startuml
class User {
  birthday: Date
  name: String
  address: String
}

class BusinessOwner {
  linkedInAccount: String
}

class PizzaRestaurant {
  zipCode: String
  address: String
  phoneNumber: String
  website: String
  openingHours: String
}

class Pizza {
  name: String
  crustStructure: String

}

  deliveryAddress: String
}

class Order {
  id: Int
  orderDate: Date
  latestDeliveryTime: Date

}

  entertainmentType: String

}

}

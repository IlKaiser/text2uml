@startuml
class User {
    name: String
    address: String
    birthday: Date
}

class BusinessOwner {
    linkedInAccount: String
}

class PizzaRestaurant {
    zipCode: String
    address: String
    phoneNumber: String
    website: String
    openingHours: String
}

class Pizza {
    name: String
    crustStructure: String
    price: Float
}

class HungryCustomer {
    deliveryAddress: String
}

class Order {
    orderID: String
    orderDateTime: Date
    latestDeliveryTime: Date
    numberOfPeople: Int
}

class EntertainmentOrder {
    entertainmentType: String
    duration: Int
}

class Entertainer {
    stageName: String
    bio: String
    pricePer30Minutes: Float
}

BusinessOwner "1..*" -- "1..1" PizzaRestaurant
PizzaRestaurant "1..*" -- "1..*" Pizza
HungryCustomer "1..*" -- "1..*" Order
Order "1..*" -- "1..*" Pizza
EntertainmentOrder "1..1" -- "1..1" Entertainer
Entertainer "0..*" -- "0..*" PizzaRestaurant

BusinessOwner --|> User

@enduml

@startuml

class AppUser{
  String UserId
  String Name
  Date Birthday
  String Address
}

class BusinessOwner{
  String LinkedInAccount
}

class HungryCustomer{
  String DeliveryAddress
}

class Entertainer{
  String StageName
  String Bio
  Double PricePer30Minutes
}

class PizzaRestaurant{
  String ZipCode
  String Address
  String PhoneNumber
  String Website
  String OpeningHours
}

class Pizza{
  String Name
  String CrustStructure
  Double Price
}

class Order{
  String OrderId
  DateTime PlacedDateTime
  DateTime LatestDeliveryTime
  Integer NumberOfPeople
}

class EntertainmentOrder{
  String EntertainmentType
  Integer DurationMinutes
}

class Availability{
  String DayOfWeek
}

AppUser <|-- BusinessOwner
AppUser <|-- HungryCustomer
AppUser <|-- Entertainer
BusinessOwner "0..*" -- "0..*" PizzaRestaurant
PizzaRestaurant "1..1" -- "0..*" Pizza
HungryCustomer "1..1" -- "0..*" Order
Order "1..*" -- "0..*" Pizza
Order <|-- EntertainmentOrder
EntertainmentOrder "1..1" -- "0..*" Entertainer
Entertainer "0..*" -- "0..*" PizzaRestaurant
(Entertainer,PizzaRestaurant) .. Availability

@enduml

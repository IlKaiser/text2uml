@startuml

class AppUser{
  String UserID
  String Name
  String Address
  Date Birthday
}

class BusinessOwner{
  String LinkedInAccount
}

class HungryCustomer{
  String DeliveryAddress
}

class Entertainer{
  String StageName
  String Bio
  Double PricePer30Minutes
}

class PizzaRestaurant{
  String RestaurantID
  String ZipCode
  String Address
  String PhoneNumber
  String Website
  String OpeningHours
}

class Pizza{
  String PizzaID
  String Name
  String CrustStructure
  Double Price
}

class Order{
  String OrderID
  Date OrderDate
  Time OrderTime
  Time LatestDeliveryTime
  Integer NumberOfPeople
}

class EntertainmentOrder{
  String EntertainmentType
  Duration Duration
}

class EntertainerRestaurantAvailability{
  String AvailableDay
}

AppUser <|-- BusinessOwner
AppUser <|-- HungryCustomer
AppUser <|-- Entertainer

BusinessOwner "0..*" -- "1..1" PizzaRestaurant
PizzaRestaurant "0..*" -- "1..1" Pizza
HungryCustomer "0..*" -- "1..*" Order
Order "1..*" -- "0..*" Pizza
Order <|-- EntertainmentOrder
EntertainmentOrder "1..1" -- "0..*" Entertainer
Entertainer "0..*" -- "0..*" PizzaRestaurant : EntertainerRestaurantAvailability

@enduml

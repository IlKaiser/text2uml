  String Name
  Date Birthday
  String Address
}

class BusinessOwner {
  String LinkedInAccount
}

class HungryCustomer {
  String DeliveryAddress
}

class Entertainer {
  String StageName
  String Bio

}

  String ZipCode
  String Address
  String PhoneNumber
  String Website
  String OpeningHours
}

  String Name
  String CrustStructure
  Double Price
}

class Order {

  DateTime PlacedAt
  DateTime LatestDeliveryTime
  Int NumberOfPeople
}

  String EntertainmentType
  Int DurationMinutes
}

class OrderItem {
  Int Quantity
}

  String DayOfWeek
}

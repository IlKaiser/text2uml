@startuml

class Customer {
    String FirstName
    String LastName
}

class Account {
    int AccountNumber
    Customer Owner
    double Balance
}

class CurrentAccount : Account {
    double OverdraftLimit
}

class SavingsAccount : Account {}

Customer "0..*" -- "1..5" Account
Account "1" -- "1" Customer
CurrentAccount <|-- Account
SavingsAccount <|-- Account

@enduml

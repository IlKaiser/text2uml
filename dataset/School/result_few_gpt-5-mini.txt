@startuml

class Person{ String FirstName, String LastName }
class Teacher{ String Email }
class Student{}
class School{ String Name }
class TeachingAssignment{ Int Hours }
class Room{ String Name }
class Classroom{ Int Capacity }
class OtherRoom{ Double SizeInSquareMeters }
class ClassGroup{ String Name }

Teacher --|> Person
Student --|> Person
Classroom --|> Room
OtherRoom --|> Room

School "1..1" -- "0..*" Student

TeachingAssignment "0..*" -- "1..1" School
School "0..1" -- "1..1" Teacher
School "1..1" -- "0..*" Room
ClassGroup "1..1" -- "10..31" Student
ClassGroup "1..1" -- "1..1" Classroom

@enduml

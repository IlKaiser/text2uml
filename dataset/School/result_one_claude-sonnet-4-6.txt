@startuml

class School {
    String Name
}
class Teacher {
    String FirstName
    String LastName
    String Email
}
class Student {
    String FirstName
    String LastName
}
class ClassGroup {
    String Name
}
class Room {
    String Name
}
class Classroom {
    String Name
    Int Capacity
}
class OtherRoom {
    String Name
    Double SizeSquareMeters
}

Room <|-- Classroom
Room <|-- OtherRoom

Student "10..31" -- "1..1" ClassGroup
ClassGroup "1..1" -- "1..1" Classroom
Student "0..*" -- "1..1" School
School "1..1" -- "1..3" Teacher
(School, Teacher) .. TeachingRelationship
School "1..1" -- "1..1" Teacher
School "0..*" -- "0..*" Room

class TeachingRelationship {
    Int Hours
}

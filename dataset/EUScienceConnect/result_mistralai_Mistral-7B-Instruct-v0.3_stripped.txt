@startuml

class Person {
    String PersonID
    String PhoneNumber
    String Institution
    String Keywords
}

class ResearchInstitution {
    String InstitutionCode
}

class Article {
    String DOI
    String Title
    List<Person> Authors
    String Position
    String Type // "Peer-Reviewed Paper" or "Technical Report"
}

class PeerReviewedPaper {
    List<Person> Reviewers
    int CitationCount
}

class TechnicalReport {
    ResearchInstitution Publisher
}

class Publisher {
    String Name
    List<Journal> Journals
    float ImpactFactor
}

class Journal {
    String Name
    ResearchInstitution Publisher
}

Person "0..*" -- "1..*" Article
Article "0..*" -- "1..*" PeerReviewedPaper
Article "0..*" -- "1..*" TechnicalReport
Publisher "1..*" -- "0..*" Journal
Journal "1..*" -- "0..*" ResearchInstitution

@enduml

@startuml

class Article {
  DOI: String
  title: String
}

class Institution {
  code: String
}

class Journal {
  name: String
  impactFactor: Float
}

class Paper {
  citationCount: Int
}

class Person {
  ID: String
  phoneNumber: String
  keywords: String
}

class Publisher {
  name: String
}

class "Technical Report"

Institution "0..*" -- "1..1" Person
Person "0..*" -- "0..*" Article
Person "0..*" -- "0..*" Paper
"Technical Report" "1..*" -- "1..1" Institution
Publisher "1..1" -- "1..*" Journal
Institution "0..*" -- "0..*" Journal
Paper "0..*" -- "1..1" Journal

Paper --|> Article
"Technical Report" --|> Article

@enduml

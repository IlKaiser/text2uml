
@startuml

class ScientificStaff {
    personID: String
    phoneNumber: String
    researchTopics: String
    institution: ResearchInstitution
}

class ResearchInstitution {
    institutionCode: String
}

class ScientificArticle {
    DOI: String
    title: String
    authors: String
    authorPosition: Int
    articleType: String
    citationCount: Int
    reviewer: ScientificStaff
}

class Publisher {
    name: String
}

class Journal {
    name: String
    impactFactor: Float
}

ScientificStaff "1..*" -- "1..1" ResearchInstitution.
ScientificStaff "1..1" -- "0..*" ScientificArticle.
ScientificArticle "1..1" -- "1..*" Publisher.
Publisher "1..*" -- "0..*" Journal.

@enduml

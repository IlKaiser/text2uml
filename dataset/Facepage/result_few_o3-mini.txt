@startuml

class User {
    String ID
    String Name
    String Email
    Date DateOfBirth
}

class Account {
    int AccountNumber
}

class BusinessAccount {
    String CompanyName
    Double MonthlyFee
}

Account <|-- PersonalAccount

    String PageName
    int NumberOfVisits
}

    Double BounceRate
    Double ClickThroughRate
    Double ConversionRate
}

Page <|-- AdvertisementPage

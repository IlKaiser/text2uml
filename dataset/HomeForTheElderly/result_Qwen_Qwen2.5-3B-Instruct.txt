class HomeForElderly{
    +String name
    +String address
    +List<Department> departments
}

class Department{
    +String name
    +List<Room> rooms
}

class Room{
    +String name
    +List<Beds> beds
}

class Beds{
    +String category
    +int quantity
    +Category category
}

class Category{
    +String name
    +int price
}

class Stay{
    +String type
    +int duration
    +HomeForElderly home
    +Invoices invoices
}

class Invoices{
    +List<Invoice> invoices
}

class Invoice{
    +Stay stay
    +int amount
    +boolean paid
}

class Person{
    +String id
    +String firstName
    +String lastName
    +List<BasicIdentificationData> basicIdentificationData
    +List<NeedsAssessment> needsAssessments
    +List<Stays> stays
}

class BasicIdentificationData{
    +String name
    +String address
    +String contactNumber
}

class NeedsAssessment{
    +Person person
    +Stay stay
    +int score
}

class Stays{
    +Person person
    +Stay stay
}

class RegistrationProcess{
    +Person person
    +List<BasicIdentificationData> basicIdentificationData
    +List<NeedsAssessment> needsAssessments
    +List<Stays> stays
}

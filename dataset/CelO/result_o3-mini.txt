@startuml
class Attendee {
  String firstName
  String lastName
  String email
  String password
  String eventResponse
}

class Organizer extends Attendee {
  String postalAddress
  String phoneNumber
}

class Event {
  DateTime startDateTime
  DateTime endDateTime
  String occasion
}

class Location {
  String name
  String address
}

class EventType {
  String name
}

class Task {
  String description
  Boolean designatedForAttendees
}

Event "1..1" -- "1..1" Location
Event "1..1" -- "1..1" EventType
EventType "1..1" -- "0..*" Task
Event "1..1" -- "1..*" Organizer
Event "1..1" -- "1..*" Attendee
Task "0..*" -- "0..*" Attendee
@enduml

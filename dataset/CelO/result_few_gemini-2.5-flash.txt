    String Password
}

    String FirstName
    String LastName
    String PostalAddress
    String PhoneNumber
}
class Attendee {
    String FirstName
    String LastName
}

    String Occasion
    DateTime StartDateTime
    DateTime EndDateTime
}

    String Name
}
class Location {
    String Name
    String Address
}
class Task {
    String Description
}

    String Status
    Boolean IsDesignatedForAttendee
}
class EventAttendee {
    String InvitationStatus
}

    String FirstName
    String EmailAddress
    String Password
}

abstract class PersonRole {}

class Organizer {
    String Address
    String PhoneNumber
}

class Attendee {}

    CompletionStatus status
}

}

}

class Location {
    String Name
    String Address
}

class Event {
    String Occasion
    Date StartTime
    Date EndTime
}

class EventType {
    String Name
}

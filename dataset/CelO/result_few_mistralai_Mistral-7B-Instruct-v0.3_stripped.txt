@startuml

class Organizer {
    String FirstName, LastName, Email, Password, Phone, Address
    String EventType
    DateTime StartDate, EndDate
    String LocationName, LocationAddress
}

class Attendee {
    String FirstName, LastName, Email
    String Status
}

class Event {
    String Type
    DateTime StartDate, EndDate
    String LocationName, LocationAddress
    List<Attendee> Attendees
    List<Task> Tasks
}

class Task {
    String Name
    Boolean IsDone
    Boolean IsAttendeeTask
}


@startuml
class GasStation {
    ID: Int
    address: String
}

class FuelPump {
    number: Int
    selfService: Boolean
}

class Fuel {
    name: String
    octaneRating: Int
}

class DailyPrice {
    date: Date
    fuel: Fuel
    gasStation: GasStation
    pricePerLiter: Float
}

class FuelPurchase {
    ID: Int
    fuelPump: FuelPump
    dailyPrice: DailyPrice
    litersDispensed: Float
}

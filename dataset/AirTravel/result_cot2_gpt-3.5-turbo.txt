}

class Flight {
    flight_number: String
    date: Date
}

class Passenger {
    name: String
    passport_number: String
}

class Ticket {
    ticket_number: String
    price: Float
    upgrade_desired: Boolean
}

class Seat {
    category: String
}

class Category {
    designation: String
    entertainment_program: Boolean
    num_seats: Int
}

class Airline {
    name: String
}

class FlightAttendant {
    name: String
    date_of_employment: Date
    is_chief_steward: Boolean
}

class Pilot {
    name: String
    date_of_employment: Date
    license: String
}

class Airport {
    name: String
    address: String
    num_runways: Int
}

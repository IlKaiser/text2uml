  year of manufacture: Int
  date of next inspection: Date
}

  name: String
}
class Airport {
  name: String
  address: String
  number of runways: Int
}
class Flight {
  flight number: Int
  date: Date
}
class FlightAttendant {
  name: String
  date of employment: Date
  is chief steward/stewardess: Boolean
}
class Passenger {
  name: String
  passport number: String
}
class PassengerAircraft {
  number of seats: Int
}
class Pilot {
  name: String
  date of employment: Date
  license: String
}
class SeatCategory {
  designation: String
  offers entertainment program: Boolean
  number of seats: Int
}
class Ticket {
  ticket number: String
  price: Float
  upgrade desired: Boolean
}

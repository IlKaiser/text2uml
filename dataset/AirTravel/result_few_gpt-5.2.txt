@startuml

class Aircraft{ 
  String Name
  String Type
  Int YearOfManufacture
  Date NextInspectionDate
}
class PassengerAircraft{
  Int NumberOfSeats
}
class Flight{
  String FlightNumber
  Date FlightDate
}
class Passenger{
  String Name
  String PassportNumber
}
class Ticket{
  String TicketNumber
  Double Price
  Boolean UpgradeDesired
}
class SeatCategory{
  String Designation
  Boolean EntertainmentProgram
  Int SeatCount
}
class Airline{
  String Name
}
class FlightAttendant{
  String Name
  Date EmploymentDate
  Boolean IsChiefSteward
}
class Pilot{
  String Name
  Date EmploymentDate
  String License
}
class Airport{
  String Name
  String Address
  Int NumberOfRunways
}

Aircraft <|-- PassengerAircraft

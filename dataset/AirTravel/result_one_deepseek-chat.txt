    String Type
    Integer YearOfManufacture
    Date NextInspectionDate
}

    String FlightNumber
    Date FlightDate
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class PassengerAircraft {
    Integer NumberOfSeats
}
class SeatCategory {
    String Designation
    Boolean HasEntertainmentProgram
    Integer NumberOfSeatsInCategory
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date EmploymentDate
    Boolean IsChief
}
class Pilot {
    String Name
    Date EmploymentDate
    String License
}
class Airport {
    String Name
    String Address
    Integer NumberOfRunways
}

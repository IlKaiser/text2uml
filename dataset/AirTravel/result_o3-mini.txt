@startuml

class Aircraft {
  String name, String type, int yearOfManufacture, Date nextInspectionDate
}

class Flight {
  String flightNumber, Date date
}

class Passenger {
  String name, String passportNumber
}

class Ticket {
  String ticketNumber, double price, boolean upgradeDesired
}

class PassengerAircraft {
  int numberOfSeats
}

class SeatCategory {
  String designation, boolean hasEntertainmentProgram, int numberOfSeats
}

class Airline {
  String name
}

class FlightAttendant {
  String name, Date dateOfEmployment, boolean isChief
}

class Pilot {
  String name, Date dateOfEmployment, String license
}

class Airport {
  String name, String address, int numberOfRunways
}

PassengerAircraft --|> Aircraft

@startuml

class Aircraft {
  String name
  String type
  Integer yearOfManufacture
  Date nextInspectionDate
}

class Flight {
  String flightNumber
  Date date
}

class Passenger {
  String name
  String passportNumber
}

class Ticket {
  String ticketNumber
  Float price
  Boolean upgradeDesired
}

class PassengerAircraft {
  Integer numberOfSeats
}

class SeatCategory {
  String designation
  Boolean hasEntertainmentProgram
  Integer numberOfSeats
}

class Airline {
  String name
}

class FlightAttendant {
  String name
  Date dateOfEmployment
  Boolean isChief
}

class Pilot {
  String name
  Date dateOfEmployment
  String license
}

class Airport {
  String name
  String address
  Integer numberOfRunways
}

Aircraft "0..*" -- "0..1" Airline
Aircraft "1..1" -- "0..*" Flight
Flight "1..1" -- "0..*" Passenger
Passenger "0..*" -- "0..*" Flight
Flight "1..1" -- "0..*" Ticket
Ticket "1..1" -- "1..1" Passenger
PassengerAircraft --|> Aircraft
PassengerAircraft "1..1" -- "0..*" SeatCategory
Ticket "1..1" -- "1..1" SeatCategory
Airline "1..1" -- "0..*" FlightAttendant
Airline "1..1" -- "0..*" Pilot
Flight "1..1" -- "0..*" FlightAttendant
Flight "1..1" -- "1..1" Pilot : captain >
Flight "1..1" -- "1..2" Pilot : co-pilot >
Aircraft "0..1" -- "0..1" Airport : home airport >
Flight "1..1" -- "1..1" Airport : departure >
Flight "1..1" -- "1..1" Airport : destination >

@enduml

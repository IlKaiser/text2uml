    String type
    Int yearOfManufacture
    Date dateOfNextInspection
}
class Flight {
    String flightNumber
    Date date
}
class Passenger {
    String name
    String passportNumber
}
class Ticket {
    String ticketNumber
    Double price
    Boolean upgradeDesired
}
class PassengerAircraft {
    Int numberOfSeats
}
class SeatCategory {
    String designation
    Boolean entertainmentProgram
    Int numberOfSeats
}
class Airline {
    String name
}
class FlightAttendant {
    String name
    Date dateOfEmployment
    Boolean chiefStewardStewardessFlag
}
class Pilot {
    String name
    Date dateOfEmployment
    String license
}
class Airport {
    String name
    String address
    Int numberOfRunways
}

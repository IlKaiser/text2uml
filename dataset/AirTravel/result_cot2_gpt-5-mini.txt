class Aircraft {
  name: String
  type: String
  yearOfManufacture: Int
  dateOfNextInspection: Date
}
class PassengerAircraft {
  numberOfSeats: Int
}
class Flight {
  flightNumber: String
  date: Date
}
class Passenger {
  name: String
  passportNumber: String
}
class Ticket {
  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}
class SeatCategory {
  designation: String
  hasEntertainmentProgram: Boolean
  numberOfSeats: Int
}
class Airline {
  name: String
}
class FlightAttendant {
  name: String
  dateOfEmployment: Date
  isChief: Boolean
}
class Pilot {
  name: String
  dateOfEmployment: Date
  license: String
}
class Airport {
  name: String
  address: String
  numberOfRunways: Int
}

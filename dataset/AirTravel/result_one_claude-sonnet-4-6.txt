@startuml

class Aircraft {
    String Name
    String Type
    Int YearOfManufacture
    Date NextInspection
}
class PassengerAircraft {
    Int NumberOfSeats
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Int NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChiefSteward
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Int NumberOfRunways
}

Aircraft <|-- PassengerAircraft
Aircraft "0..*" -- "0..1" Airline
Aircraft "0..*" -- "0..1" Airport
Airline "1..1" -- "0..*" FlightAttendant
Airline "1..1" -- "0..*" Pilot
Aircraft "1..1" -- "1..*" Flight
Passenger "0..*" -- "1..*" Flight
Passenger "1..1" -- "1..*" Ticket
Ticket "0..*" -- "1..1" Flight
Ticket "0..*" -- "1..1" SeatCategory
PassengerAircraft "1..1" -- "0..*" SeatCategory
FlightAttendant "1..*" -- "0..*" Flight
Pilot "1..3" -- "1..1" Flight
Flight "0..*" -- "1..1" Airport

@enduml

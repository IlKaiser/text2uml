@startuml
class Aircraft {
  name: String
  type: String
  year_of_manufacture: Int
  date_of_next_inspection: Date
}

class PassengerAircraft {
  number_of_seats: Int
}

class Flight {
  flight_number: String
  date: Date
}

class Passenger {
  name: String
  passport_number: String
}

class Ticket {
  ticket_number: String
  price: Float
  upgrade_desired: Boolean
}

class SeatCategory {
  designation: String
  offers_entertainment_program: Boolean
  number_of_seats: Int
}

class Airline {
  name: String
}

class FlightAttendant {
  name: String
  date_of_employment: Date
  is_chief: Boolean
}

class Pilot {
  name: String
  date_of_employment: Date
  license: String
}

class Airport {
  name: String
  address: String
  number_of_runways: Int
}

Aircraft "1..1" -- "0..*" Flight
Passenger "1..1" -- "0..*" Ticket
Flight "1..1" -- "0..*" Ticket
PassengerAircraft "1..1" -- "0..*" SeatCategory
SeatCategory "1..1" -- "0..*" Ticket
Airline "0..1" -- "0..*" Aircraft
Airline "1..1" -- "0..*" FlightAttendant
Airline "1..1" -- "0..*" Pilot
FlightAttendant "0..*" -- "0..*" Flight
Flight "1..*" -- "1..*" Pilot
Airport "1..1" -- "0..*" Flight
Airport "1..1" -- "0..*" Flight
Airport "0..1" -- "0..*" Aircraft

PassengerAircraft --|> Aircraft
@enduml

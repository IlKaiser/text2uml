```plantuml
@startuml

class Aircraft {
    name: String
    type: String
    year of manufacture: Int
    date of next inspection: Date
}

class Flight {
    flight number: String
    date: Date
}

class Passenger {
    name: String
    passport number: String
}

class Ticket {
    ticket number: String
    price: Float
    upgrade desired: Boolean
}

class SeatCategory {
    designation: String
    offers entertainment program: Boolean
    number of seats: Int
}

class Airline {
    name: String
}

class FlightAttendant {
    name: String
    date of employment: Date
    chief steward or stewardess: Boolean
}

class Pilot {
    name: String
    date of employment: Date
    license: String
}

class Airport {
    name: String
    address: String
    number of runways: Int
}

class PassengerAircraft {
    number of seats: Int
}

Aircraft "1..*" -- "1..*" Flight
Passenger "1..*" -- "1..*" Flight
Passenger "1..*" -- "1..1" Ticket
Ticket "1..1" -- "1..1" SeatCategory
Aircraft "0..1" -- "1..*" Airline
Airline "1..*" -- "1..*" FlightAttendant
Airline "1..*" -- "1..*" Pilot
FlightAttendant "1..*" -- "1..*" Flight
Pilot "1..*" -- "1..*" Flight
Aircraft "0..1" -- "1" Airport
Flight "1" -- "1" Airport
Flight "1" -- "1" Airport

PassengerAircraft --|> Aircraft

@enduml
```

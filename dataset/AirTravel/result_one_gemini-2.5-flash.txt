@startuml

class Aircraft {
    String Name
    String Type
    Integer YearOfManufacture
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class PassengerAircraft {
    Integer NumberOfSeats
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Integer NumberOfSeatsInCategory
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChiefStewardOrStewardess
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Integer NumberOfRunways
}

PassengerAircraft --|> Aircraft
Aircraft "1" -- "1..*" Flight
(Passenger,Flight) .. Ticket
PassengerAircraft "1" -- "1..*" SeatCategory
Ticket "0..*" -- "1" SeatCategory
Aircraft "0..*" -- "0..1" Airline
Airline "1" -- "1..*" FlightAttendant
Airline "1" -- "1..*" Pilot
Flight "0..*" -- "0..*" FlightAttendant
Flight "2..3" -- "0..*" Pilot
Aircraft "0..*" -- "0..1" Airport
Flight "1" -- "0..*" Airport
Flight "1" -- "0..*" Airport

@startuml

class Aircraft {
    String Name
    String Type
    Integer YearOfManufacture
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date FlightDate
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class PassengerAircraft {
    Integer NumberOfSeats
}
class SeatCategory {
    String Designation
    Boolean HasEntertainment
    Integer NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date EmploymentDate
    Boolean IsChiefSteward
}
class Pilot {
    String Name
    Date EmploymentDate
    String License
}
class Airport {
    String Name
    String Address
    Integer NumberOfRunways
}

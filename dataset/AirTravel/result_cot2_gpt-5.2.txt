class Aircraft {

  type: String
  yearOfManufacture: Int

}

  numberOfSeats: Int
}

  flightNumber: String
  date: Date
}
class Passenger {
  name: String
  passportNumber: String
}
class Ticket {
  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}
class SeatCategory {
  designation: String
  entertainmentProgram: Boolean
  numberOfSeats: Int
}
class Airline {
  name: String
}
class FlightAttendant {
  name: String
  dateOfEmployment: Date
  isChiefSteward: Boolean
}
class Pilot {
  name: String
  dateOfEmployment: Date
  license: String
}
class Airport {
  name: String
  address: String
  numberOfRunways: Int
}

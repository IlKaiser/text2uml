@startuml

class Aircraft {
    String Name
    String Type
    Int YearOfManufacture
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeWished
}
class PassengerAircraft {
    Int NumberOfSeats
}
class SeatCategory {
    String Name
    Boolean EntertainmentProgram
    Int NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChiefStewardOrStewardess
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Int NumberOfRunways
}

PassengerAircraft --|> Aircraft

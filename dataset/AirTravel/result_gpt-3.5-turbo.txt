    int flightNumber
    Date date
}

    String passportNumber
}

    int ticketNumber
    double price
    boolean upgradeDesired
}

class PassengerAircraft{
    int numberOfSeats
}

    String designation
    boolean offersEntertainment
    int numberOfSeats
}

    String name
}

class FlightAttendant{
    String name
    Date dateOfEmployment
    boolean chiefSteward
}

class Pilot{
    String name
    Date dateOfEmployment
    String license
}

class Airport{
    String name
    String address
    int numberOfRunways
}

Airport -- Aircraft : home airport
Flight -- Airport : departure airport

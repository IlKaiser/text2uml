class Aircraft{
    +String name
    +String type
    +Date yearOfManufacture
    +Date nextInspection
    +Flight[] flights
    +SeatCategory[] seatCategories
}

class Passenger{
    +String name
    +String passportNumber
    +Flight[] flights
    +Ticket ticket
}

class Flight{
    +String flightNumber
    +Date date
}

class Ticket{
    +String ticketNumber
    +int price
    +boolean upgradeDesired
}

class PassengerAircraft extends Aircraft{}

class SeatCategory{
    +String designation
    +boolean hasEntertainmentProgram
    +int numberOfSeats
}

class Airline{
    +String name
    +FlightAttendant[] flightAttendants
    +Pilot[] pilots
}

class FlightAttendant{
    +String name
    +Date dateOfEmployment
    +String role
}

class Pilot{
    +String name
    +Date dateOfEmployment
    +String license
}

class Airport{
    +String name
    +int numberOfRunways
    +Aircraft homeAircraft
    +Airport departureAirport
    +Airport destinationAirport
}

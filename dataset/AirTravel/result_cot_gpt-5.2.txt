@startuml
class Aircraft {
  name: String
  type: String
  yearOfManufacture: Int
  nextInspectionDate: Date
}

class Flight {
  flightNumber: String
  date: Date
}

class Passenger {
  name: String
  passportNumber: String
}

class Ticket {
  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}

class PassengerAircraft {
  numberOfSeats: Int
}

class SeatCategory {
  designation: String
  offersEntertainmentProgram: Boolean
  numberOfSeats: Int
}

class Airline {
  name: String
}

class FlightAttendant {
  name: String
  dateOfEmployment: Date
  isChiefSteward: Boolean
}

class Pilot {
  name: String
  dateOfEmployment: Date
  license: String
}

class Airport {
  name: String
  address: String
  numberOfRunways: Int
}

Aircraft "1..*" -- "1..1" Flight
Passenger "0..*" -- "0..*" Flight
Passenger "0..*" -- "1..1" Ticket
Flight "1..*" -- "1..1" Ticket
Ticket "1..1" -- "0..*" SeatCategory
PassengerAircraft "1..*" -- "1..1" SeatCategory
Aircraft "0..1" -- "0..*" Airline
Airline "0..*" -- "1..1" FlightAttendant
Airline "0..*" -- "1..1" Pilot
Flight "1..*" -- "0..*" FlightAttendant
Flight "1..3" -- "0..*" Pilot
Aircraft "0..1" -- "0..*" Airport
Flight "2..2" -- "0..*" Airport

PassengerAircraft --|> Aircraft
@enduml

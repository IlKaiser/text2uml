  type: String
  yearOfManufacture: Int
  nextInspectionDate: Date
}
class PassengerAircraft {
  numberOfSeats: Int
}
class Flight {
  flightNumber: String
  date: Date
}

  name: String
  passportNumber: String
}
class Ticket {
  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}
class SeatCategory {
  designation: String
  hasEntertainmentProgram: Boolean
  numberOfSeats: Int
}
class Airline {
  name: String
}
class FlightAttendant {
  name: String
  employmentDate: Date
  isChiefSteward: Boolean
}
class Pilot {
  name: String
  employmentDate: Date
  license: String
}
class Airport {
  name: String
  address: String
  numberOfRunways: Int
}
class Captain
class Copilot

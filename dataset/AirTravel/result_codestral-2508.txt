@startuml

class Aircraft{ String Name, String Type, int YearOfManufacture, Date NextInspectionDate }
class Flight{ String FlightNumber, Date FlightDate }
class Passenger{ String Name, String PassportNumber }
class Ticket{ String TicketNumber, double Price, boolean UpgradeDesired }
class PassengerAircraft{ int NumberOfSeats }
class SeatCategory{ String Designation, boolean EntertainmentProgram, int NumberOfSeats }
class Airline{ String Name }
class FlightAttendant{ String Name, Date EmploymentDate, boolean IsChiefSteward }
class Pilot{ String Name, Date EmploymentDate, String License }
class Airport{ String Name, String Address, int NumberOfRunways }

Aircraft "0..*" -- "1..*" Flight
Flight "1..*" -- "0..*" Passenger
Passenger "0..*" -- "0..*" Flight
Flight "1..*" -- "1..*" Ticket
PassengerAircraft --|> Aircraft
PassengerAircraft "1..*" -- "1..*" SeatCategory
Ticket "1..*" -- "1..1" SeatCategory
Aircraft "0..*" -- "0..1" Airline
Airline "1..*" -- "0..*" FlightAttendant
Airline "1..*" -- "0..*" Pilot
Flight "1..*" -- "1..*" FlightAttendant
Flight "1..*" -- "2..3" Pilot
Aircraft "0..*" -- "0..1" Airport
Flight "1..1" -- "1..1" Airport : departure
Flight "1..1" -- "1..1" Airport : destination

@enduml

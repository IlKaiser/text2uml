@startuml

class Aircraft {
    String Name
    String Type
    Date YearOfManufacture
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class PassengerAircraft {
    Int NumberOfSeats
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Int NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date DateOfEmployment
    Boolean IsChief
}
class Pilot {
    String Name
    Date DateOfEmployment
    String License
}
class Airport {
    String Name
    String Address
    Int NumberOfRunways
}

}

}

    Double Price
    Boolean UpgradeDesired
}

    Int NumberOfSeats
}

    String Designation

    Int NumberOfSeats
}

    String Name
}

    String Name

}

    String Name
    Date DateOfEmployment
    String License
}

    String Name
    String Address

}

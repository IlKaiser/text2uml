class Aircraft {
  - name: String
  - type: String
  - year of manufacture: Int
  - date of inspection: Date
  - flights: Flight*
  - passengers: Passenger*
  - seats: SeatCategory*
  - airline: Airline
}
class Flight {
  - flight number: String
  - date: Date
  - ticket: Ticket
  - passengers: Passenger*
}
class Passenger {
  - name: String
  - passport number: String
  - ticket: Ticket
}
class Ticket {
  - ticket number: String
  - price: Float
  - upgrade: Boolean
}
class Seat {
  - designation: String
  - seats: Int
  - category: SeatCategory
}
class SeatCategory {
  - name: String
  - seats: Int
  - entertainment program: Boolean
}
class Airline {
  - name: String
  - flight attendants: FlightAttendant*
  - pilots: Pilot*
}
class FlightAttendant {
  - name: String
  - date of employment: Date
  - is chief steward or stewardess: Boolean
}
class Pilot {
  - name: String
  - date of employment: Date
  - license: String
}
class Airport {
  - name: String
  - address: String
  - runways: Int
  - home airport: HomeAirport
  - departure airport: DepartureAirport
  - destination airport: DestinationAirport

}
class Flight {
  flightNumber: String

}
class Passenger {
  name: String
  passportNumber: String
}

  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}
class PassengerAircraft {
  numberOfSeats: Int
}
class SeatCategory {
  designation: String
  entertainmentProgram: Boolean
  numberOfSeats: Int
}
class Airline {
  name: String
}
class FlightAttendant {
  name: String
  dateOfEmployment: Date
  isChiefSteward: Boolean
}
class Pilot {
  name: String
  dateOfEmployment: Date
  license: String
}
class Airport {
  name: String
  address: String
  numberOfRunways: Int
}

@startuml

class Aircraft {
    String Name
    String Type
    Integer YearOfManufacture
    Date NextInspectionDate
}
class Flight {
    String FlightNumber
    Date Date
}
class Passenger {
    String Name
    String PassportNumber
}
class Ticket {
    String TicketNumber
    Double Price
    Boolean UpgradeDesired
}
class PassengerAircraft {
    Integer NumberOfSeats
}
class SeatCategory {
    String Designation
    Boolean EntertainmentProgram
    Integer NumberOfSeats
}
class Airline {
    String Name
}
class FlightAttendant {
    String Name
    Date EmploymentDate
    Boolean IsChief
}
class Pilot {
    String Name
    Date EmploymentDate
    String License
}
class Airport {
    String Name
    String Address
    Integer NumberOfRunways
}

Aircraft "1" -- "0..*" Flight
Flight "1" -- "0..*" Passenger
Passenger "0..*" -- "0..*" Flight
Flight "1" -- "0..*" Ticket
PassengerAircraft "1" -- "0..*" SeatCategory
Ticket "1" -- "1" SeatCategory
Aircraft "0..*" -- "1" Airline
Airline "1" -- "0..*" FlightAttendant
Airline "1" -- "0..*" Pilot
Flight "1" -- "0..*" FlightAttendant
Flight "1" -- "1" Pilot
Flight "1" -- "1" Airport : departure
Flight "1" -- "1" Airport : destination
Aircraft "0..1" -- "1" Airport : home

PassengerAircraft --|> Aircraft

```plantuml
@startuml

class Aircraft {
  name: String
  type: String
  year_of_manufacture: Int
  date_of_next_inspection: Date
}

class Flight {
  flight_number: String
  date: Date
}

class Passenger {
  name: String
  passport_number: String
}

class Ticket {
  ticket_number: String
  price: Float
  upgrade_desired: Boolean
}

class PassengerAircraft {
  number_of_seats: Int
}

class SeatCategory {
  designation: String
  entertainment_program: Boolean
  number_of_seats: Int
}

class Airline {
  name: String
}

class FlightAttendant {
  name: String
  date_of_employment: Date
  is_chief: Boolean
}

class Pilot {
  name: String
  date_of_employment: Date
  license: String
}

class Airport {
  name: String
  address: String
  number_of_runways: Int
}

Aircraft "1..*" -- "1..*" Flight
Flight "1..*" -- "1..*" Passenger
Flight "1..*" -- "1..1" Ticket
PassengerAircraft "1..*" -- "1..*" SeatCategory
Aircraft "0..1" -- "1..*" Airline
Airline "1..*" -- "1..*" FlightAttendant
Airline "1..*" -- "1..*" Pilot
Flight "1..*" -- "1..*" FlightAttendant
Flight "1..1" -- "1..*" Pilot
Aircraft "0..1" -- "1..*" Airport
Flight "1..1" -- "1..1" Airport
PassengerAircraft --|> Aircraft

@enduml
```

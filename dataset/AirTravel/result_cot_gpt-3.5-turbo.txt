
@startuml
class Aircraft {
    name: String
    type: String
    year_of_manufacture: Date
    date_of_next_inspection: Date
}

class Flight {
    flight_number: Int
    date: Date
}

class Passenger {
    name: String
    passport_number: String
}

class Ticket {
    ticket_number: Int
    price: Float
    upgrade_desired: Boolean
}

class SeatCategory {
    designation: String
    entertainment_program: Boolean
    num_seats: Int
}

class Airline {
    name: String
}

class FlightAttendant {
    name: String
    date_of_employment: Date
    chief_steward: Boolean
    stewardess: Boolean
}

class Pilot {
    name: String
    date_of_employment: Date
    license: String
}

class Airport {
    name: String
    address: String
    num_runways: Int
}

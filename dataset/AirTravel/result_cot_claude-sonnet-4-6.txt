@startuml
class Aircraft {
  name: String
  type: String
  yearOfManufacture: Int
  dateOfNextInspection: Date
}
class Flight {
  flightNumber: String
  date: Date
}
class Passenger {
  name: String
  passportNumber: String
}
class Ticket {
  ticketNumber: String
  price: Float
  upgradeDesired: Boolean
}
class PassengerAircraft {
  numberOfSeats: Int
}
class SeatCategory {
  designation: String
  entertainmentProgram: Boolean
  numberOfSeats: Int
}
class Airline {
  name: String
}
class FlightAttendant {
  name: String
  dateOfEmployment: Date
  isChiefSteward: Boolean
}
class Pilot {
  name: String
  dateOfEmployment: Date
  license: String
}
class Airport {
  name: String
  address: String
  numberOfRunways: Int
}
Aircraft "0..*" -- "1..1" Flight
Flight "1..*" -- "1..*" Passenger
Passenger "1..1" -- "1..*" Ticket
Ticket "1..1" -- "1..*" SeatCategory
PassengerAircraft "1..1" -- "1..*" SeatCategory
Aircraft "0..*" -- "0..1" Airline
Airline "1..1" -- "1..*" FlightAttendant
Airline "1..1" -- "1..*" Pilot
Flight "1..*" -- "1..*" FlightAttendant
Flight "1..*" -- "2..3" Pilot
Aircraft "0..1" -- "0..*" Airport
Flight "1..*" -- "1..1" Airport
PassengerAircraft --|> Aircraft
@enduml

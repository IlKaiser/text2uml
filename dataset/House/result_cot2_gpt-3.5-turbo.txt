
@startuml
class House {
  address: String
  buildingMaterial: String
}

class Basement {
  size: Int
  name: String
  type: String
  humidityLevel: Int
}

class SemiDetachedHouse {
  address: String
  buildingMaterial: String
  hasOwnGarden: Boolean
  numWindows: Int
  carportType: String
  garageInfo: String
}

class Company {
  name: String
  address: String
}

class Job {
  numHours: Int
  agreedPrice: Float
}

House "1..1" -- "0..1" Basement
Basement --|> EarthBasement
Basement --|> ConcreteBasement
House "1..1" -- "0..*" Company
Company "1..*" -- "1..*" Job
House "1..1" -- "0..1" SemiDetachedHouse
SemiDetachedHouse --|> House
SemiDetachedHouse "1..1" -- "1..1" Carport
SemiDetachedHouse "1..1" -- "1..1" Garage
@enduml

class Traveller {

}

    Date ArrivalDate

    int NumberOfRooms
    String RoomType
    int HotelRating

    String Name
    String Address
    String City
    int Rating
}

class Room {
    String Type

    Date AvailabilityStartDate
    Date AvailabilityEndDate

}

    double Price
    String CityArea
    int HotelRating
    String KeyPreferences

}

    String Name
}

}

class Traveller {
    String Name
    String BillingCompany
    String BillingAddress
    String Preferences
    Double ReliabilityRating
}

class Hotel {
    String Name
    String City
    String Address
    String Chain
}

class RoomType {
    String Type
    String Description
    Integer Capacity
    Double PricePerNight
}

class Booking {
    String BookingIdentifier
    Date ArrivalDate
    Date DepartureDate
    Integer NumRooms
    String RoomType
    Integer MinHotelStars
    Double BudgetPerNight
    String City
    String Status
    String CancellationDeadline
    Boolean IsPrePaid
    Boolean PaidAtHotel
    Date ConfirmationDeadline
}

class Offer {
    Double OfferPrice
    Date ValidUntil
    String Status
    String Notes
}

class CreditCard {
    String CardNumber
    String CardHolderName
    Date ExpiryDate
    String CardType
}

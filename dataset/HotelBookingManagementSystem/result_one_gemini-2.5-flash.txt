@startuml

class Traveller {
    String Name
    String BillingInformation
    String CompanyName
    String CompanyAddress
    String TravelPreferences
    Double ReliabilityRating
}
class Hotel {
    String Name
    String Address
    String City
    Integer Rating
}
class HotelChain {
    String Name
}
class RoomTypeAvailability {
    String Type
    Date StartDate
    Date EndDate
    Integer AvailableRooms
}
abstract class Booking {
    String City
    Date ArrivalDate
    Date DepartureDate
    Integer NumberOfRooms
    String RoomType
}
class PreliminaryBooking {
    String UniqueBookingIdentifier
}
class FinalizedBooking {
    String UniqueBookingIdentifier
    String CreditCardInformation
    String PaymentType
    String Status
    Date CancellationDeadline
    Double ChargedCancellationFee
}
class Offer {
    Double Price
    String CityArea
    Integer HotelRating
    String KeyPreferences
}
class SpecialOffer {
    Double Price
    String CityArea
    Integer HotelRating
    String KeyPreferences
    String OfferDetails
}
class Reimbursement {
    Double Amount
    Date ReimbursementDate
}
class Compensation {
    Double Amount
    String Description
}

Booking <|-- PreliminaryBooking
Booking <|-- FinalizedBooking

class Student {

    String Email
}
class Tutor {
    String Name
    String Email
    String BankAccount
}
class Subject {
    String Name
}
class TutoringOffer {
    String ExpertiseLevel
    Double HourlyPrice
}
class Availability {
    String DayOfWeek
    Time StartTime
    Time EndTime
}
class TutoringRequest {
    String Level
    Date TargetDate
    Time TargetTime
}
class TutoringSession {
    Date SessionDate
    Time SessionTime
    String Status
}
class Payment {
    Double Amount
    String PaymentMethod
}
class Discount {
    Double Percentage
}
class Award {
    Date Month
    String Subject
}

@startuml

class User {
  String name
  String email
}

class Tutor {
  String bankAccount
}

User <|-- Tutor

class Availability {
  String day
  String startTime
  String endTime
}

class TutoringOffer {
  String subject
  String level
  double hourlyPrice
}

class TutoringRequest {
  String subject
  String level
  Date targetDateTime
}

class TutoringSession {
  Date scheduledDateTime
  String status
}

class Payment {
  String method
}

class TutorOfTheMonthAward {
  String subject
  Date awardMonth
}

Tutor "1..1" -- "0..*" Availability
Tutor "1..1" -- "0..*" TutoringOffer

User "1..1" -- "0..*" TutoringRequest
Tutor "1..1" -- "0..*" TutoringRequest

User "1..1" -- "0..*" TutoringSession
Tutor "1..1" -- "0..*" TutoringSession

TutoringSession "1..1" -- "1..1" Payment

Tutor "1..1" -- "0..*" TutorOfTheMonthAward

@enduml

@startuml
' Classes
class WebPortal { 
  String Version
  String URL 
}

class Building { 
  String UniqueID
  String Name
  String EnergyClass
  String Address
  int YearOfConstruction 
}

class AdditionalEntry { 
  String Name
  boolean IsPublic
}

class TextEntry {
  String Text
}

class NumericEntry {
  float Value
  String Unit
}

class Image {
  String FileName
  boolean IsProfilePhoto
}

class User { 
  String UserName
  String Email
  String Password
}

class AdministrativeUser { 
  String IDCardNumber
}

class Comment { 
  String CommentText
}

' Inheritance for additional entry types
AdditionalEntry <|-- TextEntry
AdditionalEntry <|-- NumericEntry

' Inheritance for user types
User <|-- AdministrativeUser

' Relations
WebPortal "1..1" -- "0..*" Building

Building "1..1" -- "0..*" AdditionalEntry
Building "1..1" -- "0..*" Image
User "1..1" -- "0..*" Comment
Building "1..1" -- "0..*" Comment
@enduml

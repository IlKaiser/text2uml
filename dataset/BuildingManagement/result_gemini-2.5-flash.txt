@startuml

class WebPortal{
    String VersionNumber
    String URLAddress
}

class Building{
    String UniqueID
    String Name
    String EnergyClass
    String Address
    int YearOfConstruction
}

class User{
    String UserName
    String EmailAddress
    String Password
    boolean IsActivated
}

abstract class Entry{
    String Name
    boolean IsPubliclyReleased
}

class TextEntry{
    String Content
}

class NumericEntry{
    float Value
    String Unit
}

class Image{
    String FileName
    boolean IsProfilePhoto
}

class Administrator{
    String IDCardNumber
}

class Comment{
    String Content
    Date Timestamp
}

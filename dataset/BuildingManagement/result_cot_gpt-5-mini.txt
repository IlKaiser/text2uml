@startuml
class WebPortal {
  version: String
  url: String
}

class Building {
  id: Int
  name: String
  energyClass: String
  address: String
  yearOfConstruction: Int
}

class User {
  username: String
  email: String
  password: String
}

class Administrator {
  idCardNumber: String
}

class Entry {
  name: String
  isPublic: Boolean
}

class TextEntry {
  text: String
}

class NumericEntry {
  value: Float
  unit: String
}

class Image {
  fileName: String
}

class Comment {
  text: String
}

WebPortal "0..*" -- "1..1" Building

Building "0..*" -- "1..1" Image
Entry "1..1" -- "0..*" User
Comment "1..1" -- "0..*" Building

Administrator --|> User
TextEntry --|> Entry

@enduml

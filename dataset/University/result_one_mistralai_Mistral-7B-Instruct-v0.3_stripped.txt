@startuml

class University {
    String Name
    int TotalEmployees
}

class Faculty {
    String Name
    University University
    Employee Dean
}

class Institute {
    String Name
    String Address
    Faculty Faculty
}

class Employee {
    String SocialSecurityNumber
    String Name
    String Email
}

class ResearchStaff extends Employee {
    Institute Institute
    String FieldOfResearch
}

class AdministrativeStaff extends Employee {
}

class ResearchAssistant extends ResearchStaff {
    int HoursInProjects
    Project[] Projects
}

class Lecturer extends ResearchAssistant {
    Course[] Courses
}

class Project {
    String Name
    Date StartDate
    Date EndDate
}

class Course {
    int ID
    String Name
    int WeeklyDuration
}

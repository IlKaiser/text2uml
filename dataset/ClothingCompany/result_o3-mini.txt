class Product {
  String Name
  String Description
  double NetPrice
  String Manufacturer
}

class Country {
  double VatRate
  String Currency
  double ConversionRate
}

class Representative {
  String FirstName
  String LastName
  String SocialSecurityNumber
  String Telephone
  String Email
}

class Customer {
  String Name
  String Address
  String Telephone
  String Email
}

class Order {
  String OrderID
  int Quantity

  Date DeliveryDate
}

@startuml
class Company {
    - name: String
    - products: Product[]
}
class Product {
    - name: String
    - description: String
    - price: Float
    - company: Company
    - country: Country
    - vatRate: Rate
    - currency: Currency
    - conversionRate: Float
}
class Country {
    - name: String
    - vatRate: Rate
    - currency: Currency
    - conversionRate: Float
}
class Rate {
    - name: String
    - value: Float
}
class Currency {
    - symbol: String
    - exchangeRate: Float
}
class CustomerData {
    - name: String
    - address: Address
    - telephoneNumber: String
    - email: String
    - countries: Country[]
}
class Address {
    - street: String
    - city: String
    - state: String
    - zipCode: String
}
class Order {
    - id: String
    - quantity: Int
    - orderDate: Date
    - deliveryDate: Date
    - product: Product
    - representative: Representative
}
class Representative {
    - firstName: String
    - lastName: String
    - socialSecurityNumber: String
    - telephoneNumber: String
    - email: String
    - country: Country
}
class OrderItem {
    - id: String
    - quantity: Int
    - order: Order
}
@enduml

@startuml

class Product {
    String Name
    String Description
    Double NetPrice
    String Manufacturer
    String CountryOfOrigin
}

class Vat {
    String Country
    Double VatRate
    String Currency
    Double ConversionRateToEuro
}

class Country {
    String Name
    List<Product> Products
}

class Representative {
    String FirstName
    String LastName
    String SocialSecurityNumber
    String TelephoneNumber
    String Email
    Country Country
}

class Customer {
    String Name
    String Address
    String TelephoneNumber
    String Email
    List<Country> Countries
}

class Order {
    String Id
    Product Product
    Int Quantity
    Date OrderDate
    Date DesiredDeliveryDate
    Representative Representative
}

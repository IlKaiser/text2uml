class Product{ String UniqueName, String Description, Decimal NetPrice }
class Country{ String Name, Decimal VATRate, String Currency, Decimal EuroConversionRate }
class Manufacturer{ String Name }
class Representative{ String FirstName, String LastName, String SocialSecurityNumber, String TelephoneNumber, String EmailAddress }
class Customer{ String Name, String Address, String TelephoneNumber, String EmailAddress, String[] Countries }
class Order{ String OrderID, Integer Quantity, Date OrderDate, Date DeliveryDate }

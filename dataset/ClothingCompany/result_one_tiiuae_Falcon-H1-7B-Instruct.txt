@startuml

class Customer {
    String Name
    String Address
    String TelephoneNumber
    String Email
    "0..*" String Countries
}
class Product {
    String UniqueName
    String Description
    Double NetPrice
    String Manufacturer
    String Country
}
class Country {
    String Name
    Double VATRate
    String Currency
    Double ConversionRateToEuro
}
class Representative {
    String FirstName
    String LastName
    String SocialSecurityNumber
    String TelephoneNumber
    String Email
    "1" Country Country
}
class Order {
    String OrderID
    Date OrderDate
    Date DesiredDeliveryDate
    "1" Product Product
    Int Quantity
}

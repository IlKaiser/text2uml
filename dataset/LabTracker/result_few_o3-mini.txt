@startuml

class Doctor {
    int PractitionerNumber
    String Signature
    String FullName
    String Address
    String PhoneNumber
}

class Patient {
    String HealthNumber
    String FirstName
    String LastName
    Date DateOfBirth
    String Address
    String PhoneNumber
}

class Requisition {
    Date ValidFrom
}

class Test {
    String Type
    String Duration
    String Result
    String AppointmentType
}

class RepetitionPattern {
    int RepeatCount
    String Interval
}

class Report {
    String Content
}

class Lab {
    String Address
    String RegistrationNumber
    String BusinessHours
    Double ChangeCancellationFee
}

class Appointment {
    String ConfirmationNumber
    Date Date
    String StartTime
    String EndTime
}

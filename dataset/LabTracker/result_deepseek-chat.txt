@startuml

class Doctor {
  String fullName
  String address
  String phoneNumber
  Integer practitionerNumber
  Image signature
}

class Patient {
  String healthNumber
  String firstName
  String lastName
  Date dateOfBirth
  String address
  String phoneNumber
}

class Requisition {
  Date validFrom
  String repetitionInterval
  Integer repetitionCount
}

class Test {
  String name
  Integer duration
  Boolean requiresAppointment
  Boolean isWalkInOnly
  Boolean requiresSampleDropOff
}

class TestGroup {
  String groupName
}

class Appointment {
  String confirmationNumber
  Date appointmentDate
  Time startTime
  Time endTime
  Boolean isCancelledOrChanged
}

class Lab {
  String name
  String address
  String registrationNumber
  Time dailyStartTime
  Time dailyEndTime
  Float changeCancellationFee
}

class TestResult {
  String result
  String report
}

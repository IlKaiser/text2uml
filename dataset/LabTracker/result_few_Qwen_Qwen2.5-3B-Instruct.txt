    String Signature
    String FullName
    String Address
    String PhoneNumber

}
class Patient {
    String HealthNumber
    String FirstName
    String LastName
    Date DateOfBirth

    String PhoneNumber
}

    String DoctorPractitionerNumber
    String DoctorSignature
    String DoctorFullName
    String DoctorAddress
    String DoctorPhoneNumber

    Patient Patient
    List<Test> Tests
}

    String Type
    String Duration
}

    String ConfirmationNumber
    Date Date
    Time StartTime
    Time EndTime
    String LabName
    String LabRegistrationNumber
}

    String Type
}

    String Type

}

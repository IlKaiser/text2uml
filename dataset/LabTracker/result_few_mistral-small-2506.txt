@startuml

class Doctor {
    String PractitionerNumber
    String FullName
    String Address
    String PhoneNumber
    String DigitalSignature
}

class Patient {
    String HealthNumber
    String FirstName
    String LastName
    Date DateOfBirth
    String Address
    String PhoneNumber
}

class Requisition {
    Date ValidFrom
    String Status
}

class Test {
    String Name
    String Group
    String Type
    Int Duration
    Boolean RequiresAppointment
    Boolean IsWalkInOnly
    Boolean RequiresSampleDropOff
}

class Appointment {
    String ConfirmationNumber
    Date Date
    Time StartTime
    Time EndTime
    String Status
}

class Lab {
    String RegistrationNumber
    String Address
    Time StartTime
    Time EndTime
    Double ChangeCancellationFee
}

class TestResult {
    Boolean Result
    String Report
}

class RepetitionPattern {
    Int NumberOfTimes
    String Interval
}

@startuml

class Customer {
  String VATNumber
  String HeadOfficeAddress
}

class DeliveryAddress {
  String Address
}

class Order {
  Date OrderDate
  Boolean IsFullyShipped
  Boolean IsFullyPaid
  Date PaidDate
}

class Invoice {
  Date InvoiceDate
  Boolean IsPaid
}

class OrderLine {
  Integer QuantityOrdered
  Boolean IsShipped
}

class Shipment {
  String Status
  Date PackedDate
  Date ShippedDate
  Date ReceivedDate
}

class Product {
  String Name
  Integer CurrentStock
}

class ShipmentLine {
  Integer QuantityShipped
}

@startuml

class SensorController {
    String Id
}
class Apartment {
    String Address
    String Email
}
class Inhabitant {
    String Username
    String Password
    Boolean IsAdmin
}
class Room {
    String Description
}
class Sensor {
    String VendorId
    String Description
    String Unit
    Double Threshold
}
class TemperatureSensor {
    Double MinValue
    Double MaxValue
}
class PowerSensor {
    Double TotalConsumption
}
class Measurement {
    Date Timestamp
    Double Value
}
class EMonitor {
    String Id
}

PowerSensor --|> Sensor

@startuml
class SensorController

class Sensor {
  vendorId: String
  description: String
  unit: String
  threshold: Float
  room: String
}

class TemperatureSensor {
  min: Float
  max: Float
}

class PowerSensor {
  counterReading: Float
}

class Measurement {
  timestamp: Date
  value: Float
}

class Apartment {
  address: String
  email: String
}

class Room {
  description: String
}

class User {
  username: String
  password: String
  isAdmin: Boolean
}

class EMonitor

TemperatureSensor --|> Sensor
PowerSensor --|> Sensor

SensorController "1..1" -- "1..1" Apartment
Apartment "1..*" -- "1..1" Room
Room "0..*" -- "1..1" Sensor
Sensor "0..*" -- "1..1" Measurement
User "1..1" -- "1..*" SensorController
User "1..*" -- "1..*" Sensor
Sensor "1..1" -- "1..*" EMonitor
Apartment "1..1" -- "1..1" EMonitor
@enduml

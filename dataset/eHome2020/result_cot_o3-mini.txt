@startuml
class Apartment {
  address: String
  email: String
}

class Inhabitant {
  username: String
  password: String
  isAdmin: Boolean
}

class Room {
  description: String
}

class Sensor {
  vendorID: String
  description: String
  unit: String
}

class TemperatureSensor {
  min: Float
  max: Float
}

class PowerSensor {
  counterReading: Float
}

class MeasuredValue {
  timestamp: Date
  value: Float
}

Apartment "1..*" -- "1..1" Inhabitant
Apartment "2..*" -- "1..1" Room
Sensor "1..1" -- "0..*" Room
Sensor "0..*" -- "1..1" MeasuredValue
TemperatureSensor --|> Sensor
PowerSensor --|> Sensor
@enduml

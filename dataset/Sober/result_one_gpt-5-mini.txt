@startuml

class Customer { String Name, String ContactInfo, Integer RideShareBookingCount }
class Owner { String Name, String ContactInfo }
class Car { String LicensePlate, String Make, String Model, Boolean IsCompanyOwned }
class Accident { Date Date, String Location, Double DamageAmount }
class Ride { Date PickupTime, Date DropoffTime, Double Duration, Double Distance, Integer NumberOfPassengers }
class RideHail { Double Fee, String RequestType, Integer MaxPassengers }
class RideShare { Double UpfrontFee, Integer MaxPassengers }

RideHail --|> Ride
RideShare --|> Ride

Customer "1" -- "0..*" RideHail
Customer "1..10" -- "0..*" RideShare
Owner "0..1" -- "0..*" Car

Car "1" -- "0..*" RideShare
Car "1" -- "0..*" Accident

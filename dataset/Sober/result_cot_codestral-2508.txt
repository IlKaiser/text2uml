```plantuml
@startuml

class Car {
    ownerName:String
    ownerContact:String
}

class Customer {
    name:String
    contact:String
}

class RideHailService {
    pickupTime:Date
    dropoffTime:Date
    pickupLocation:String
    dropoffLocation:String
    rideDuration:Float
    distance:Float
    numPassengers:Int
    fee:Float
    requestType:String
    leadCustomerName:String
    leadCustomerContact:String
}

class RideShareService {
    pickupTime:Date
    dropoffTime:Date
    pickupLocation:String
    dropoffLocation:String
    rideDuration:Float
    distance:Float
    customerNames:String[]
    customerContacts:String[]
    negotiatedFee:Float
}

class Accident {
    date:Date
    location:String
    damageAmount:Float
}

class Sober
class Request
class TreePlanting

Sober "1..*" -- "1..*" Car
Sober "1..*" -- "1..*" Customer
Sober "1..*" -- "1..*" RideHailService
Sober "1..*" -- "1..*" RideShareService
Sober "1..*" -- "0..*" Request
Sober "1..*" -- "0..*" Accident
Sober "1..*" -- "0..*" TreePlanting
Car "1..*" -- "0..1" Customer
Customer "1..*" -- "0..*" RideHailService
Customer "1..*" -- "0..*" RideShareService
RideHailService "1..1" -- "1..*" Request
RideShareService "1..1" -- "1..*" Request
Car "1..*" -- "0..*" Accident

RideHailService --|> RideShareService

@enduml
```

@startuml

class Car {
}

class SelfDrivingCar {
}

class RegisteredCar {
}

class CarOwner {
    String Name
    String ContactInfo
}

class Customer {
    String Name
    String ContactInfo
}

abstract class TaxiService {
    DateTime PickUpTime
    DateTime DropOffTime
    String PickUpLocation
    String DropOffLocation
    Float RideDuration
    Float Distance
    Integer NumberOfPassengers
}

class RideHailingService {
    Float Fee
    String RequestType
}

class RideSharingService {
    Float NegotiatedFee
}

class Accident {
    Date AccidentDate
    String Location
    Float DamageAmount
}

Car <|-- SelfDrivingCar
Car <|-- RegisteredCar
TaxiService <|-- RideHailingService
TaxiService <|-- RideSharingService

CarOwner "1" -- "0..*" RegisteredCar
Car "1" -- "0..*" TaxiService
Customer "0..*" -- "1" RideHailingService
Customer "0..*" -- "1..*" RideSharingService
Car "1" -- "0..*" Accident

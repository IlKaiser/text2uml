@startuml

class Car {
    String RegistrationNumber
    String Make
    String Model
    Boolean IsSoberOwned
}
class CarOwner {
    String Name
    String Email
    String PhoneNumber
}
class Customer {
    String Name
    String Email
    String PhoneNumber
    Int RideSharingUsesCount
}
abstract class Ride {
    DateTime TimeOfPickup
    DateTime TimeOfDropoff
    String PickupLocation
    String DropoffLocation
    Int RideDurationMinutes
    Double Distance
}
class RideHailingRide {
    Int NumberOfPassengers
    Double Fee
    String RequestType
}
class RideSharingRide {
    Double NegotiatedFee
}
class Accident {
    Date AccidentDate
    String Location
    Double DamageAmount
}

CarOwner "1" -- "0..*" Car
Car "1" -- "0..*" Ride
Customer "1" -- "0..*" RideHailingRide
Customer "0..*" -- "0..*" RideSharingRide
Car "1" -- "0..*" Accident
Ride <|-- RideHailingRide
Ride <|-- RideSharingRide

@enduml

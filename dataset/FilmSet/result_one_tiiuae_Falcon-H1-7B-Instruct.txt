@startuml

class Employee {
    String Name
    String SocialSecurityNumber
}
class CreativeEmployee --|> Employee {
    boolean IsOscarWinner
}
class Actor --|> CreativeEmployee {
    String Education
}
class Director --|> CreativeEmployee {
    String AssistantName
}
class Technician --|> Employee {
    String FieldOfActivity
}
class FilmSet {
    String Location
}
class Film {
    String Title
    int YearOfRelease
    String Genre
}
class Screenwriter {
    String MostSuccessfulFilm
}
class Screenplay {
    String Title
    String Version
    Date Date
    int NumberOfScenes
    String Plot
}
class Novel {
    String Title
    String Author
}
class Concept {}

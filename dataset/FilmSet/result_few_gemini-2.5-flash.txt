class Employee {

}
class CreativeEmployee {
    Boolean OscarWinner
}
class Director {
    String AssistantName
}
class Actor {
    String Education
}
class Technician {
    String FieldOfActivity
}
class FilmSet {
    String Location
}
class Film {
    String Title
    Int ReleaseYear
    String Genre
}

    String MostSuccessfulFilmTitle
}

    String Title
    String Version

}

    String Author
}
class ConceptBasis {}

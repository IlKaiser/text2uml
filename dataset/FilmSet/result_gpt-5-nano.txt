@startuml

class Employee { String Name, String SocialSecurityNumber }

abstract class CreativeEmployee { boolean HasOscar }
class Actor { String EducationCompleted }
class Director { String AssistantName }
class Technician { String FieldOfActivity }

Actor --|> CreativeEmployee
Director --|> CreativeEmployee

class Screenwriter { String Name, String MostSuccessfulFilm }

abstract class Source { }
class Novel extends Source { String Title, String Author }
class Concept extends Source { String Description }

Novel --|> Source
Concept --|> Source

class Screenplay { String Title, String Version, String Date, int NumberOfScenes, String Plot }

class Film { String Title, int YearOfRelease }
class FilmSet { String Location }
class Genre { String Name }

Screenwriter "1" -- "0..*" Screenplay
Screenplay "1" -- "1" Film
Screenplay "1" -- "1" Source
Novel --|> Source
Concept --|> Source

FilmSet "0..*" -- "0..*" Film
Film "1" -- "1" Genre
Genre "0..*" -- "1" Film
Actor "0..*" -- "1" Screenplay
Director "1" -- "1" Screenplay
FilmSet "0..*" -- "0..*" Employee

@enduml

@startuml

class Location{
    +String address
}

class Driver{
    +String firstName
    +String lastName
    -Location location
    -Vehicle vehicle
}

class Vehicle{
    +String vehicleNumber
    +VehicleType vehicleType
    -Driver driver
}

enum VehicleType {
    Truck
    Car
    Van
}

class Order{
    +String origin
    +String destination
    -Driver driver
    +Date executionDate
}

Order "1" -- "1..1" Driver
Order "1" -- "1" Vehicle
Driver "1" -- "1..1" Vehicle
@enduml
